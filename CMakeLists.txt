cmake_minimum_required(VERSION 3.20)
project(hiyui_terminal_app LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
if(NOT TARGET hiyui::hiyui)
    if(NOT MSVC)
        enable_language(ASM)
    endif()
    if(HIYUI_SDK_PATH AND HIYUI_SOURCE_DIR)
        message(FATAL_ERROR "HIYUI_SDK_PATH and HIYUI_SOURCE_DIR are mutually exclusive")
    endif()
    if(HIYUI_SDK_PATH)
        get_filename_component(HIYUI_SDK_PATH "${HIYUI_SDK_PATH}" ABSOLUTE
            BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
        if(hiyui_DIR)
            get_filename_component(_hiyui_config_dir "${hiyui_DIR}" ABSOLUTE)
            file(RELATIVE_PATH _hiyui_config_relative "${HIYUI_SDK_PATH}" "${_hiyui_config_dir}")
            if(IS_ABSOLUTE "${_hiyui_config_relative}" OR _hiyui_config_relative MATCHES "^\\.\\./")
                set(hiyui_DIR "hiyui_DIR-NOTFOUND")
            endif()
        endif()
        find_package(hiyui CONFIG REQUIRED PATHS "${HIYUI_SDK_PATH}" NO_DEFAULT_PATH)
        get_target_property(_hiyui_sdk_prefix hiyui::hiyui HIYUI_SDK_PREFIX)
        file(RELATIVE_PATH _hiyui_sdk_relative "${HIYUI_SDK_PATH}" "${_hiyui_sdk_prefix}")
        if(NOT _hiyui_sdk_relative STREQUAL "" AND NOT _hiyui_sdk_relative STREQUAL ".")
            message(FATAL_ERROR "HIYUI_SDK_PATH does not match the selected SDK: ${_hiyui_sdk_prefix}")
        endif()
    else()
        if(NOT HIYUI_SOURCE_DIR)
            set(HIYUI_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../..")
        endif()
        get_filename_component(HIYUI_SOURCE_DIR "${HIYUI_SOURCE_DIR}" ABSOLUTE
            BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
        add_subdirectory("${HIYUI_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/hiyui")
    endif()
endif()

if(ANDROID OR EMSCRIPTEN OR NOT CMAKE_SYSTEM_NAME MATCHES "^(Windows|Linux|Darwin)$")
    message(FATAL_ERROR "The local terminal application requires a desktop platform")
endif()
if(DEFINED HIYUI_BACKEND AND NOT HIYUI_BACKEND STREQUAL "SDL")
    message(FATAL_ERROR "The local terminal application requires the SDL backend")
endif()

find_package(Threads REQUIRED)
add_library(hiyui_terminal_session STATIC
    src/terminal_session.cpp src/share_protocol.cpp src/shared_session.cpp)
target_include_directories(hiyui_terminal_session PUBLIC src PRIVATE third_party/cpp-httplib)
target_link_libraries(hiyui_terminal_session PUBLIC hiyui::hiyui Threads::Threads)
target_compile_definitions(hiyui_terminal_session PRIVATE
    CPPHTTPLIB_WEBSOCKET_MAX_PAYLOAD_LENGTH=8388608
    CPPHTTPLIB_WEBSOCKET_CLOSE_TIMEOUT_SECOND=1)
if(MSVC)
    target_compile_options(hiyui_terminal_session PRIVATE /utf-8 /bigobj)
endif()
add_executable(hiyui_terminal src/main.cpp)
target_link_libraries(hiyui_terminal PRIVATE hiyui_terminal_session)
if(WIN32)
    target_sources(hiyui_terminal_session PRIVATE src/shell_session_win.cpp)
    target_compile_definitions(hiyui_terminal_session PRIVATE _WIN32_WINNT=0x0A00 NOMINMAX)
    target_link_libraries(hiyui_terminal_session PRIVATE ws2_32 bcrypt crypt32)
    target_compile_definitions(hiyui_terminal PRIVATE _WIN32_WINNT=0x0A00)
    set_target_properties(hiyui_terminal PROPERTIES WIN32_EXECUTABLE TRUE)
    if(MINGW)
        target_link_options(hiyui_terminal PRIVATE -municode)
    endif()
else()
    target_sources(hiyui_terminal_session PRIVATE src/shell_session_posix.cpp)
    find_library(HIYUI_TERMINAL_UTIL util)
    if(HIYUI_TERMINAL_UTIL)
        target_link_libraries(hiyui_terminal_session PRIVATE ${HIYUI_TERMINAL_UTIL})
    endif()
endif()
hiyui_setup_application(hiyui_terminal)
if(MSVC)
    get_target_property(_terminal_runtime hiyui_terminal MSVC_RUNTIME_LIBRARY)
    set_property(TARGET hiyui_terminal_session PROPERTY MSVC_RUNTIME_LIBRARY "${_terminal_runtime}")
endif()
hiyui_install_application(hiyui_terminal LICENSE_FILES
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/cpp-httplib/LICENSE")

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    enable_testing()
endif()
add_executable(test_terminal_sharing tests/test_sharing.cpp)
target_link_libraries(test_terminal_sharing PRIVATE hiyui_terminal_session)
target_include_directories(test_terminal_sharing PRIVATE third_party/cpp-httplib)
target_compile_definitions(test_terminal_sharing PRIVATE
    CPPHTTPLIB_WEBSOCKET_MAX_PAYLOAD_LENGTH=8388608
    CPPHTTPLIB_WEBSOCKET_CLOSE_TIMEOUT_SECOND=1)
if(MSVC)
    target_compile_options(test_terminal_sharing PRIVATE /utf-8 /bigobj)
    set_property(TARGET test_terminal_sharing PROPERTY MSVC_RUNTIME_LIBRARY "${_terminal_runtime}")
endif()
add_test(NAME terminal_sharing COMMAND test_terminal_sharing)
set_tests_properties(terminal_sharing PROPERTIES TIMEOUT 60)
add_test(NAME terminal_tunnel COMMAND test_terminal_sharing --tunnel-tests)
set_tests_properties(terminal_tunnel PROPERTIES TIMEOUT 60)

add_executable(test_terminal_layout tests/test_layout.cpp)
target_include_directories(test_terminal_layout PRIVATE src)
target_link_libraries(test_terminal_layout PRIVATE hiyui::hiyui)
hiyui_setup_application(test_terminal_layout)
add_test(NAME terminal_layout COMMAND test_terminal_layout)
set_tests_properties(terminal_layout PROPERTIES TIMEOUT 30)
